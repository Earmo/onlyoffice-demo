    在 `insertRemoteImage` 函数之后追加：
    ```javascript
    async function handleInsertHtml({ html }) {
      // 评审 Issue #4：显式双重守卫 — bridge 实例存在 AND bridgeReady.value 握手完成
      // 仅检查 bridge !== null 不够：bridge 对象可能已创建但握手尚未完成
      if (!onlyofficeBridge || !bridgeReady.value) {
        writeBackStore.status = "error";
        writeBackStore.errorMsg = "文档连接未就绪，请等待编辑器完全加载后重试。";
        return;
      }
      try {
        await onlyofficeBridge.insertHtml(html);
        // 评审 Issue #1：通过 store 通知 Workbench，不在 shell 层调用 ElMessage
        writeBackStore.status = "success";
      } catch (error) {
        const msg = error instanceof Error ? error.message : "写入文档失败，请重试。";
        writeBackStore.status = "error";
        writeBackStore.errorMsg = msg;
      }
    }
    ```

    注意：`handleInsertHtml` 不调用任何 `ElMessage`；所有用户反馈由 EditorAiWorkbench 的 store watch 处理。

    ```javascript
    describe("handleInsertHtml", () => {
      it("bridge 未就绪时应将 store.status 设为 error", async () => {
        const wrapper = mountEditorShell({ bridgeReady: false });
        const store = useWriteBackStore();

        await wrapper.vm.handleInsertHtml({ html: "<p>test</p>" });

        expect(store.status).toBe("error");
        expect(store.errorMsg).toMatch(/未就绪/);
      });

      it("bridge.insertHtml 成功时应将 store.status 设为 success", async () => {
        const mockBridge = { insertHtml: vi.fn().mockResolvedValue({ success: true }) };
        const wrapper = mountEditorShell({ bridgeReady: true, bridge: mockBridge });
        const store = useWriteBackStore();

        await wrapper.vm.handleInsertHtml({ html: "<p>hello</p>" });

        expect(mockBridge.insertHtml).toHaveBeenCalledWith("<p>hello</p>");
        expect(store.status).toBe("success");
      });

      it("bridge.insertHtml 抛出异常时应将 store.status 设为 error 并透传 errorMsg", async () => {
        const mockBridge = {
          insertHtml: vi.fn().mockRejectedValue(new Error("PasteHtml timeout"))
        };
        const wrapper = mountEditorShell({ bridgeReady: true, bridge: mockBridge });
        const store = useWriteBackStore();

        await wrapper.vm.handleInsertHtml({ html: "<p>fail</p>" });

        expect(store.status).toBe("error");
        expect(store.errorMsg).toBe("PasteHtml timeout");
      });
    });
    ```
